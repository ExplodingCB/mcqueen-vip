cmake_minimum_required(VERSION 3.16)
project(mcq_track LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(mcq_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_subdirectory(core)
add_library(track_component SHARED src/track_server.cpp)
target_compile_options(track_component PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(track_component PUBLIC mcq_track_core yaml-cpp rclcpp::rclcpp rclcpp_components::component ${mcq_msgs_TARGETS})
rclcpp_components_register_node(track_component PLUGIN "mcq_track::TrackServer" EXECUTABLE track_server)
install(TARGETS mcq_track_core track_component ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY core/include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(mcq_track_core)
ament_package()
