cmake_minimum_required(VERSION 3.16)
project(mcq_track_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mcq_track_core SHARED src/track.cpp)
target_include_directories(mcq_track_core PUBLIC include)
target_compile_options(mcq_track_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(MCQ_TRACK_BUILD_TESTS "Build track core host tests" ON)
if(MCQ_TRACK_BUILD_TESTS)
  enable_testing()
  add_executable(track_probe test/track_probe.cpp)
  target_link_libraries(track_probe PRIVATE mcq_track_core)
  add_executable(test_track_core test/test_track.cpp)
  target_link_libraries(test_track_core PRIVATE mcq_track_core)
  add_test(NAME track_core COMMAND test_track_core)
endif()
